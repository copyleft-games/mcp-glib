* MCP Server Guide
This guide explains how to create MCP servers using mcp-glib.

** Overview
An MCP server exposes capabilities to AI clients:

- *Tools* - Functions the AI can call
- *Resources* - Data the AI can read
- *Prompts* - Templates for AI interactions

** Basic Server Structure
#+begin_src C
#include <mcp/mcp.h>

static GMainLoop *main_loop = NULL;

static void
on_client_disconnected (McpServer *server, gpointer user_data)
{
    g_main_loop_quit (main_loop);
}

int
main (int argc, char *argv[])
{
    g_autoptr(McpServer) server = NULL;
    g_autoptr(McpStdioTransport) transport = NULL;

    main_loop = g_main_loop_new (NULL, FALSE);

    /* Create server with name and version */
    server = mcp_server_new ("my-server", "1.0.0");

    /* Optional: Set instructions for the AI */
    mcp_server_set_instructions (server, "Instructions for the AI...");

    /* Connect disconnect signal to quit */
    g_signal_connect (server, "client-disconnected",
                      G_CALLBACK (on_client_disconnected), NULL);

    /* Add tools, resources, prompts here */

    /* Create stdio transport and start */
    transport = mcp_stdio_transport_new ();
    mcp_server_set_transport (server, MCP_TRANSPORT (transport));
    mcp_server_start_async (server, NULL, NULL, NULL);

    /* Run until client disconnects */
    g_main_loop_run (main_loop);
    g_main_loop_unref (main_loop);

    return 0;
}
#+end_src

** Adding Tools
Tools are functions that AI clients can call.

*** Defining a Tool
#+begin_src C
/* Tool handler function */
static McpToolResult *
greet_handler (McpServer   *server,
               const gchar *name,
               JsonObject  *arguments,
               gpointer     user_data)
{
    McpToolResult *result;
    const gchar *person = NULL;
    g_autofree gchar *greeting = NULL;

    /* Get arguments */
    if (arguments != NULL && json_object_has_member (arguments, "name"))
    {
        person = json_object_get_string_member (arguments, "name");
    }

    if (person == NULL)
    {
        /* Return error result */
        result = mcp_tool_result_new (TRUE);  /* TRUE = is_error */
        mcp_tool_result_add_text (result, "Missing required argument 'name'");
        return result;
    }

    /* Create greeting */
    greeting = g_strdup_printf ("Hello, %s!", person);

    /* Return success result */
    result = mcp_tool_result_new (FALSE);  /* FALSE = success */
    mcp_tool_result_add_text (result, greeting);

    return result;
}

/* In main() or setup function */
{
    g_autoptr(McpTool) tool = NULL;
    g_autoptr(JsonBuilder) builder = NULL;
    g_autoptr(JsonNode) schema = NULL;

    /* Create tool */
    tool = mcp_tool_new ("greet", "Greet a person by name");

    /* Define input schema */
    builder = json_builder_new ();
    json_builder_begin_object (builder);
    json_builder_set_member_name (builder, "type");
    json_builder_add_string_value (builder, "object");
    json_builder_set_member_name (builder, "properties");
    json_builder_begin_object (builder);
    json_builder_set_member_name (builder, "name");
    json_builder_begin_object (builder);
    json_builder_set_member_name (builder, "type");
    json_builder_add_string_value (builder, "string");
    json_builder_set_member_name (builder, "description");
    json_builder_add_string_value (builder, "The name of the person to greet");
    json_builder_end_object (builder);
    json_builder_end_object (builder);
    json_builder_set_member_name (builder, "required");
    json_builder_begin_array (builder);
    json_builder_add_string_value (builder, "name");
    json_builder_end_array (builder);
    json_builder_end_object (builder);

    schema = json_builder_get_root (builder);
    mcp_tool_set_input_schema (tool, schema);

    /* Register with server */
    mcp_server_add_tool (server, tool, greet_handler, NULL, NULL);
}
#+end_src

** Adding Resources
Resources are data that AI clients can read.

*** Static Resources
#+begin_src C
/* Resource handler */
static GList *
readme_handler (McpServer   *server,
                const gchar *uri,
                gpointer     user_data)
{
    McpResourceContents *contents;
    const gchar *text = "# My Server\n\nThis is the readme.";

    contents = mcp_resource_contents_new_text (uri, text, "text/markdown");

    return g_list_append (NULL, contents);
}

/* In main() */
{
    g_autoptr(McpResource) resource = NULL;

    resource = mcp_resource_new ("file:///readme", "README");
    mcp_resource_set_description (resource, "Server documentation");
    mcp_resource_set_mime_type (resource, "text/markdown");

    mcp_server_add_resource (server, resource, readme_handler, NULL, NULL);
}
#+end_src

*** Resource Templates
Resource templates allow dynamic URIs with parameters:

#+begin_src C
/* Template handler - handles any matching URI */
static GList *
file_handler (McpServer   *server,
              const gchar *uri,
              gpointer     user_data)
{
    McpResourceContents *contents;
    g_autofree gchar *path = NULL;
    g_autofree gchar *file_contents = NULL;
    g_autoptr(GError) error = NULL;

    /* Extract path from URI (e.g., "file:///docs/intro.md" -> "docs/intro.md") */
    if (!g_str_has_prefix (uri, "file:///"))
    {
        return NULL;
    }
    path = g_strdup (uri + 8);

    /* Read file */
    if (!g_file_get_contents (path, &file_contents, NULL, &error))
    {
        g_warning ("Failed to read %s: %s", path, error->message);
        return NULL;
    }

    contents = mcp_resource_contents_new_text (uri, file_contents, "text/plain");

    return g_list_append (NULL, contents);
}

/* In main() */
{
    g_autoptr(McpResourceTemplate) template = NULL;

    template = mcp_resource_template_new ("file:///{path}", "File Access");
    mcp_resource_template_set_description (template, "Access files by path");
    mcp_resource_template_set_mime_type (template, "text/plain");

    mcp_server_add_resource_template (server, template, file_handler, NULL, NULL);
}
#+end_src

*** Binary Resources
#+begin_src C
static GList *
image_handler (McpServer   *server,
               const gchar *uri,
               gpointer     user_data)
{
    McpResourceContents *contents;
    g_autofree gchar *data = NULL;
    gsize length;
    g_autofree gchar *base64 = NULL;

    /* Read binary file */
    g_file_get_contents ("image.png", &data, &length, NULL);

    /* Encode as base64 */
    base64 = g_base64_encode ((guchar *) data, length);

    /* Create blob contents */
    contents = mcp_resource_contents_new_blob (uri, base64, "image/png");

    return g_list_append (NULL, contents);
}
#+end_src

** Adding Prompts
Prompts are templates for AI interactions.

#+begin_src C
/* Prompt handler */
static McpPromptResult *
greeting_handler (McpServer   *server,
                  const gchar *name,
                  GHashTable  *arguments,
                  gpointer     user_data)
{
    McpPromptResult *result;
    McpPromptMessage *message;
    const gchar *user_name = NULL;
    g_autofree gchar *text = NULL;

    /* Get arguments from hash table */
    if (arguments != NULL)
    {
        user_name = g_hash_table_lookup (arguments, "name");
    }

    if (user_name == NULL)
    {
        user_name = "friend";
    }

    /* Create result with description */
    result = mcp_prompt_result_new ("A friendly greeting");

    /* Add message(s) */
    text = g_strdup_printf ("Hello, %s! How can I help you today?", user_name);
    message = mcp_prompt_message_new (MCP_ROLE_ASSISTANT);
    mcp_prompt_message_add_text (message, text);
    mcp_prompt_result_add_message (result, message);
    mcp_prompt_message_unref (message);

    return result;
}

/* In main() */
{
    g_autoptr(McpPrompt) prompt = NULL;

    prompt = mcp_prompt_new ("greeting", "Generate a friendly greeting");

    /* Add optional argument */
    mcp_prompt_add_argument_full (prompt, "name", "Name to greet", FALSE);

    mcp_server_add_prompt (server, prompt, greeting_handler, NULL, NULL);
}
#+end_src

** JSON-RPC ids keep their type

JSON-RPC 2.0 requires a response id to be *identical* to the request's --- the
type as well as the value. A strict client compares both.

mcp-glib exposes ids as strings, because that is one type for what the wire
allows to be a string or a number. The originating type is carried alongside:

- =mcp_request_get_id_is_number()= --- whether the id arrived as a JSON number.
- =mcp_response_set_id_is_number()= / =mcp_error_response_set_id_is_number()=
  --- answer in kind.

=McpServer= does this for you; it matters only if you build responses yourself.

#+begin_src c
  response = mcp_response_new (mcp_request_get_id (request), result);
  mcp_response_set_id_is_number (response,
                                 mcp_request_get_id_is_number (request));
#+end_src

A string id that merely *looks* numeric still comes back quoted --- that is the
other half of the same rule, and it is why the flag exists instead of a
"does this parse as an integer" test at serialisation time.

*This is not cosmetic.* Answering =\{"id": 0\}= with =\{"id": "0"\}= makes a
strict client reject the handshake, and the error it prints reads as nonsense
--- =conflict initialized response id: expected 0, got 0= --- because both sides
render the same and only one is a number. The visible symptom is a client with
no tools and nothing in any log explaining why. Lenient clients never notice,
so this can survive a long time against one client and fail immediately against
another.

Anything embedding an older mcp-glib keeps the old behaviour until it is
rebuilt.

** What a peer sends is data, not a promise

Every field in an inbound frame arrives from somebody else, so its
*presence* and its *type* are two different questions.

=json_object_has_member()= answers the first.
=json_object_get_string_member()= on a number answers NULL without a
warning, and =json_object_get_object_member()= on a string logs a
Json-CRITICAL and answers NULL. A caller that checked only presence spends
that NULL: =g_hash_table_contains()= on a =g_str_hash= table dereferences
it, so ={"method":"tools/call","params":{"name":123}}= was a remote
segfault, and =clientInfo= with a non-string =name= reached
=mcp_implementation_new()=' precondition.

Handlers read every member through the type-checked accessors in
=src/mcp-json-util.h=, and a member of the wrong type is answered with
=MCP_ERROR_INVALID_PARAMS= rather than being read as absent-but-present.

*** Ids and methods

=mcp_message_new_from_json()= sets a =GError= on *every* NULL return,
because both message handlers print =error->message= on that path. The
cases that used to return NULL silently:

- ="id": null= on a request --- legal JSON-RPC, forbidden by MCP, and the
  spelling that reached =mcp_request_new()= with a NULL id
- an id that is an object, an array or a boolean
- a non-string =method=
- a non-string =message= inside an =error= object

An error response is the one place a null id is legal --- a parse error
has no request to name --- and that still parses.

*** An empty result is an empty object

=json_node_new (JSON_NODE_OBJECT)= answers =JSON_NODE_HOLDS_OBJECT()= with
TRUE and =json_node_get_object()= with NULL. It is not ={}=; it is an
object-shaped hole, and serialising one walks into json-glib with a NULL
object and dies. =ping=, =resources/subscribe=, =resources/unsubscribe=
and =tasks/cancel= all built their replies that way, which no test could
see because both mock transports pass =JsonNode=\s in memory and never
serialise. Build an empty result with =mcp_json_new_empty_object()=.

*** Cancellation

=notifications/cancelled= names a request the *client* sent, in the
client's id space. The server's own outbound requests (sampling, roots)
are numbered in a counter of their own that also starts at 1, so looking
the id up there let a client abort server work it knew nothing about.
Inbound requests are dispatched and answered synchronously, so by the time
a cancellation arrives there is nothing in flight to stop; a long-running
call is a Task, and its cancellation is =tasks/cancel=.

** Local server hardening

The server accepts =initialize= only while CONNECTING, requires string =protocolVersion= and =clientInfo.name/version= plus an object =capabilities=, and accepts =notifications/initialized= only while INITIALIZING. Application methods require READY; =ping= remains available during initialization. A repeated initialize request returns an invalid-request error without replacing negotiated state. Stopping a server completes any pending start operation with a connection-closed error. Start callbacks can run immediately while completing a GTask; internal ownership is released before invoking them.

Tool schemas describe the contract but are not a general JSON Schema validator. A supplied =arguments= member must be an object; absent arguments remain allowed. Application handlers must validate nested fields, required properties, permitted values, and resource limits before mutation.

=McpUnixSocketServer= refuses any existing socket pathname, including stale sockets and regular files. Callers must choose an absent path inside an owned private directory, normally mode 0700. Shutdown removes the entry only when its device/inode identity and socket type still match the bound entry. The library does not authenticate peer credentials; directory access is the caller's responsibility. Listener setup cleans up its own bound entry after a listen/registration failure.

The =max-sessions= property defaults to 32 (range 1–4096); excess connections close before allocating an MCP session. =handshake-timeout-ms= defaults to 10000 (range 1–600000); idle clients are disconnected unless initialization completes in time. Properties affect newly accepted connections. A session leaves the tracked list before =session-closed= is emitted, and its server remains valid during the signal. Shutdown from creation or close callbacks is supported.
